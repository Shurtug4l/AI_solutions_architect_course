# 1. CREATE A VIRTUAL ENVIRONMENT
# -----------------------------

# Check the Python version
python --version

# Create a project directory
mkdir my_project
cd my_project

# Create the virtual environment
python -m venv env

# 2. ACTIVATE AND DEACTIVATE THE VIRTUAL ENVIRONMENT
# --------------------------------------------

# On Windows:
# .\env\Scripts\activate

# On macOS and Linux:
source env/bin/activate

# Check the active environment (should show the environment path)
which python
which pip

# To deactivate the environment:
deactivate

# 3. INSTALL PACKAGES
# -------------------

# Reactivate the environment if needed
source env/bin/activate  # (macOS/Linux)
# .\env\Scripts\activate  # (Windows)

# Install a few sample packages
pip install numpy
pip install pandas
pip install matplotlib

# Install a specific version
pip install requests==2.25.1

# 4. PACKAGE VERSIONS AND REQUIREMENTS.TXT
# -----------------------------------

# List all installed packages
pip list

# Create the requirements.txt file
pip freeze > requirements.txt

# Show the file content
cat requirements.txt

# How to install from requirements.txt in a fresh environment
# pip install -r requirements.txt

# 5. DELETE A VIRTUAL ENVIRONMENT
# -------------------------------

# Deactivate the environment first
deactivate

# On Windows:
# delete the folder

# On macOS and Linux:
rm -rf env
