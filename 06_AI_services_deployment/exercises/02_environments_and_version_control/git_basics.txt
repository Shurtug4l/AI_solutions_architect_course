# 1. Initial configuration
# -----------------------------

# Configure your name and email
git config --global user.name "Your Name"
git config --global user.email "your.email@example.com"

# Check the configuration
git config --list

# 2. Initialize a repository
# -----------------------------

# Create a new directory
mkdir my-project
cd my-project

# Initialize the repository
git init

# Check the status
git status

# Add specific files
git add filename.txt

# Add all files
git add .

# Remove a file from the staging area
git reset filename.txt

# 3. Commit, pull and push
# -----------------------------

# Create a commit with a message
git commit -m "First commit: added main file"

# Amend the last commit
git commit --amend -m "New message for the last commit"

# Show the commit history
git log
git log --oneline  # Compact version

# Add a remote repository
git remote add origin https://github.com/username/repository.git

# List the remote repositories
git remote -v

# Fetch changes from the remote repository
git pull origin main

# Upload your changes to the remote repository
git push origin main

# Push of the first commit
git push -u origin main  # -u sets up tracking

# 5. Working on a branch
# -----------------------------

# List all branches
git branch

# Create a new branch
git branch feature-login

# Switch to another branch
git checkout feature-login

# Delete a branch
git branch -d branch-name
